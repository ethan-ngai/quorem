class Solution {
public int[] levelSpreads(TreeNode root){if(root==null)return new int[0];ArrayDeque<TreeNode> q=new ArrayDeque<>();q.add(root);ArrayList<Integer>a=new ArrayList<>();while(!q.isEmpty()){int size=q.size(),lo=Integer.MAX_VALUE,hi=Integer.MIN_VALUE;for(int i=0;i<size;i++){TreeNode n=q.remove();lo=Math.min(lo,n.val);hi=Math.max(hi,n.val);if(n.left!=null)q.add(n.left);if(n.right!=null)q.add(n.right);}a.add(hi-lo);}return a.stream().mapToInt(Integer::intValue).toArray();}
}
