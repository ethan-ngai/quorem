class Solution {
public String quotaWindow(String text,String quota){
    int[] chars=text.codePoints().toArray(), requested=quota.codePoints().toArray();
    if(requested.length==0)return "";
    HashMap<Integer,Integer> need=new HashMap<>();
    for(int c:requested)need.put(c,need.getOrDefault(c,0)+1);
    int missing=requested.length,left=0,start=0,best=Integer.MAX_VALUE;
    for(int right=0;right<chars.length;right++){
        int c=chars[right]; if(need.getOrDefault(c,0)>0)missing--;need.put(c,need.getOrDefault(c,0)-1);
        while(missing==0){
            if(right-left+1<best){best=right-left+1;start=left;}
            int old=chars[left++];need.put(old,need.getOrDefault(old,0)+1);if(need.get(old)>0)missing++;
        }
    }
    return best==Integer.MAX_VALUE?"":new String(chars,start,best);
}
}
