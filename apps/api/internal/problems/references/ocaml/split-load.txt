let splitLoad loads groups = let lo=ref (List.fold_left max 0 loads) and hi=ref (List.fold_left (+) 0 loads) in while !lo < !hi do let mid=(!lo + !hi)/2 in let used=ref 1 and total=ref 0 in List.iter (fun x -> if !total+x>mid then (incr used;total:=0);total:= !total+x) loads; if !used<=groups then hi:=mid else lo:=mid+1 done; !lo
