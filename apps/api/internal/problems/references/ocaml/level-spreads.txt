let levelSpreads (root:tree_node option) = let rec loop acc level = match level with [] -> List.rev acc | _ -> let lo=List.fold_left (fun v (n:tree_node) -> min v n.value) max_int level and hi=List.fold_left (fun v (n:tree_node) -> max v n.value) min_int level in let next=List.concat_map (fun (n:tree_node) -> List.filter_map Fun.id [n.left;n.right]) level in loop ((hi-lo)::acc) next in match root with None -> [] | Some n -> loop [] [n]
